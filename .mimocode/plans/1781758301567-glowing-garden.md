# 修复扇形绘制bug和角度方向问题

## 问题描述
1. 扇形绘制出来是"三角形 + 交叉圆弧"，不是正确的扇形
2. 第二步完成时（确定半径和起始角度），应该是一条线段，但显示为交叉图形
3. 调整角度控制点时，鼠标移动方向和实际角度变化方向相反

## 根本原因

### 问题1：绘制路径错误
- `arcTo` 参数格式错误：使用了1/16度格式，但arcTo需要度
- `lineTo` + `arcTo` 导致重复画线到弧起点，产生交叉
- span_angle计算逻辑错误

### 问题2：第二步显示异常
- 初始化时 `_end_angle = 0`，但 `_start_angle` 已经是某个角度
- 绘制时 span = 0 - start_angle，可能是负数或很大的角度
- **解决方案**：第二步时将 `_end_angle` 设置为与 `_start_angle` 相同

### 问题3：角度方向相反
- 使用 `atan2(-dy, dx)` 计算角度，这假设Y轴向上（数学坐标系）
- 但Qt的Y轴向下，导致鼠标向下移动时角度增加（应该是减少）

## 修复方案

### 修改文件1：`test/roi/shapes/sector.py`

#### 统一角度系统：使用Qt坐标系（Y轴向下）

```python
# 角度定义（Qt坐标系）：
# - 0° 在3点钟方向
# - 顺时针为正（因为Y轴向下）
# - 范围：0° 到 360°

# 计算角度：使用 atan2(dy, dx) 而不是 atan2(-dy, dx)
# 计算点：使用 cos 和 sin（不用-sin）
```

#### 修复 `_build_sector_path` 方法

```python
def _build_sector_path(self) -> QPainterPath:
    path = QPainterPath()
    path.moveTo(self._center)
    
    start_rad = math.radians(self._start_angle)
    
    start_point = QPointF(self._center.x() + self._radius * math.cos(start_rad),
                          self._center.y() + self._radius * math.sin(start_rad))
    path.lineTo(start_point)
    
    rect = QRectF(self._center.x() - self._radius, self._center.y() - self._radius,
                  self._radius * 2, self._radius * 2)
    
    # Qt arcTo: 0° at 3 o'clock, clockwise positive
    span = self._end_angle - self._start_angle
    span = span % 360
    if span <= 0:
        span += 360
    
    path.arcTo(rect, self._start_angle, span)
    path.lineTo(self._center)
    path.closeSubpath()
    
    return path
```

#### 修复 `contains_point` 方法

```python
def contains_point(self, point: QPointF) -> bool:
    dx = point.x() - self._center.x()
    dy = point.y() - self._center.y()
    dist = math.sqrt(dx * dx + dy * dy)
    if dist > self._radius:
        return False
    
    # Qt coordinate system: Y-axis down
    angle = math.degrees(math.atan2(dy, dx))
    if angle < 0:
        angle += 360
    
    start = self._start_angle % 360
    end = self._end_angle % 360
    
    if start <= end:
        return start <= angle <= end
    else:
        return angle >= start or angle <= end
```

#### 修复 `control_points` 方法

```python
def control_points(self) -> list:
    start_rad = math.radians(self._start_angle)
    end_rad = math.radians(self._end_angle)
    
    return [
        QPointF(self._center),  # Center
        QPointF(self._center.x() + self._radius, self._center.y()),  # 0° point
        QPointF(self._center.x() + self._radius * math.cos(start_rad),
                self._center.y() + self._radius * math.sin(start_rad)),  # Start
        QPointF(self._center.x() + self._radius * math.cos(end_rad),
                self._center.y() + self._radius * math.sin(end_rad)),  # End
    ]
```

#### 修复 `resize_by_control` 方法

```python
def resize_by_control(self, index: int, pos: QPointF):
    if index == 0:  # Center
        self._center = QPointF(pos)
    elif index == 1:  # Radius
        self._radius = max(distance(pos, self._center), 5.0)
    elif index == 2:  # Start angle
        dx = pos.x() - self._center.x()
        dy = pos.y() - self._center.y()
        self._start_angle = math.degrees(math.atan2(dy, dx))  # 不用-dy
    elif index == 3:  # End angle
        dx = pos.x() - self._center.x()
        dy = pos.y() - self._center.y()
        self._end_angle = math.degrees(math.atan2(dy, dx))  # 不用-dy
```

### 修改文件2：`test/roi/canvas.py`

#### 修复 `_add_sector_point` 方法

```python
def _add_sector_point(self, pos: QPointF):
    if not self._drawing_shape or not isinstance(self._drawing_shape, SectorShape):
        return
    
    if self._mode == InteractionMode.DRAWING_SECTOR_STEP1:
        self._drawing_shape._center = QPointF(pos)
        self._mode = InteractionMode.DRAWING_SECTOR_STEP2
        self.update()
    elif self._mode == InteractionMode.DRAWING_SECTOR_STEP2:
        dx = pos.x() - self._drawing_shape._center.x()
        dy = pos.y() - self._drawing_shape._center.y()
        self._drawing_shape._radius = max(distance(pos, self._drawing_shape._center), 5.0)
        self._drawing_shape._start_angle = math.degrees(math.atan2(dy, dx))
        self._drawing_shape._end_angle = self._drawing_shape._start_angle  # 关键：第二步时end=start
        self._mode = InteractionMode.DRAWING_SECTOR_STEP3
        self.update()
    elif self._mode == InteractionMode.DRAWING_SECTOR_STEP3:
        dx = pos.x() - self._drawing_shape._center.x()
        dy = pos.y() - self._drawing_shape._center.y()
        self._drawing_shape._end_angle = math.degrees(math.atan2(dy, dx))
        min_size = 5
        if self._drawing_shape._radius >= min_size:
            self.add_shape(self._drawing_shape)
        self._drawing_shape = None
        self._mode = InteractionMode.IDLE
        self.update()
```

#### 修复 `_update_sector_drawing` 方法

```python
def _update_sector_drawing(self, pos: QPointF):
    if not self._drawing_shape or not isinstance(self._drawing_shape, SectorShape):
        return
    
    dx = pos.x() - self._drawing_shape._center.x()
    dy = pos.y() - self._drawing_shape._center.y()
    
    if self._mode == InteractionMode.DRAWING_SECTOR_STEP2:
        self._drawing_shape._radius = max(distance(pos, self._drawing_shape._center), 5.0)
        self._drawing_shape._start_angle = math.degrees(math.atan2(dy, dx))
        self._drawing_shape._end_angle = self._drawing_shape._start_angle  # 关键：保持同步
    elif self._mode == InteractionMode.DRAWING_SECTOR_STEP3:
        self._drawing_shape._end_angle = math.degrees(math.atan2(dy, dx))
    self.update()
```

## 关键修改总结

| 位置 | 修改前 | 修改后 |
|------|--------|--------|
| atan2参数 | `atan2(-dy, dx)` | `atan2(dy, dx)` |
| sin计算 | `-sin(angle)` | `sin(angle)` |
| arcTo参数 | `start_angle_qt / 16` | `start_angle`（直接使用） |
| span计算 | `if span < 0: span += 360` | `span = span % 360` |
| 第二步end_angle | 保持默认值0 | 设置为与start_angle相同 |
