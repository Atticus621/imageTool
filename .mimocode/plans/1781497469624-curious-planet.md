# 计划：相机系统架构设计

## 目标
设计一个解耦合、可扩展的相机系统，支持电脑内置摄像头和罗技摄像头，能够精细调整相机参数，未来可扩展支持多种工业相机。

## 需求分析

### 当前需求
- 支持电脑内置摄像头（USB/UVC协议）
- 支持罗技摄像头（USB/UVC协议）
- 精细调整相机参数（曝光时间、增益、白平衡、分辨率等）

### 未来扩展需求
- 支持工业相机（GigE、USB3 Vision等）
- 支持IP网络摄像头（RTSP/HTTP）
- 支持多相机同时使用
- 支持相机触发模式

## 架构设计原则

1. **单一职责原则**：每个类只负责一个功能
2. **开闭原则**：对扩展开放，对修改关闭
3. **依赖倒置原则**：依赖抽象而非具体实现
4. **接口隔离原则**：客户端不应依赖它不需要的接口

## 核心接口设计

### 1. ICamera 接口

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum
from typing import Any, Dict, List, Optional, Tuple
import numpy as np


class ParameterType(Enum):
    """参数类型枚举"""
    INTEGER = "integer"
    FLOAT = "float"
    BOOLEAN = "boolean"
    STRING = "string"
    ENUM = "enum"


@dataclass
class ParameterInfo:
    """参数信息"""
    name: str
    display_name: str
    param_type: ParameterType
    min_value: Optional[float] = None
    max_value: Optional[float] = None
    step: Optional[float] = None
    default_value: Any = None
    options: Optional[List[str]] = None  # 用于ENUM类型
    unit: Optional[str] = None
    description: Optional[str] = None


@dataclass
class CameraInfo:
    """相机信息"""
    camera_id: str
    name: str
    model: str
    serial_number: str
    interface_type: str  # "usb", "gige", "usb3vision", etc.
    driver_name: str


class ICamera(ABC):
    """相机抽象接口"""
    
    @abstractmethod
    def connect(self) -> bool:
        """连接相机"""
        pass
    
    @abstractmethod
    def disconnect(self) -> None:
        """断开连接"""
        pass
    
    @abstractmethod
    def is_connected(self) -> bool:
        """检查是否已连接"""
        pass
    
    @abstractmethod
    def get_camera_info(self) -> CameraInfo:
        """获取相机信息"""
        pass
    
    @abstractmethod
    def start_capture(self) -> bool:
        """开始采集"""
        pass
    
    @abstractmethod
    def stop_capture(self) -> None:
        """停止采集"""
        pass
    
    @abstractmethod
    def is_capturing(self) -> bool:
        """检查是否正在采集"""
        pass
    
    @abstractmethod
    def grab_frame(self, timeout_ms: int = 5000) -> Optional[np.ndarray]:
        """获取一帧图像
        
        Args:
            timeout_ms: 超时时间（毫秒）
        
        Returns:
            numpy数组，失败返回None
        """
        pass
    
    @abstractmethod
    def list_parameters(self) -> List[str]:
        """列出所有可用参数名称"""
        pass
    
    @abstractmethod
    def get_parameter_info(self, param_name: str) -> ParameterInfo:
        """获取参数信息"""
        pass
    
    @abstractmethod
    def get_parameter(self, param_name: str) -> Any:
        """获取参数值"""
        pass
    
    @abstractmethod
    def set_parameter(self, param_name: str, value: Any) -> bool:
        """设置参数值
        
        Args:
            param_name: 参数名称
            value: 参数值
        
        Returns:
            是否设置成功
        """
        pass
    
    def get_parameter_range(self, param_name: str) -> Tuple[Any, Any]:
        """获取参数范围（最小值，最大值）"""
        info = self.get_parameter_info(param_name)
        return (info.min_value, info.max_value)
    
    def get_all_parameters(self) -> Dict[str, Any]:
        """获取所有参数的当前值"""
        return {name: self.get_parameter(name) for name in self.list_parameters()}
    
    def set_multiple_parameters(self, params: Dict[str, Any]) -> bool:
        """批量设置参数"""
        success = True
        for name, value in params.items():
            if not self.set_parameter(name, value):
                success = False
        return success
```

class CameraManager:
    """相机管理器"""
    
    def __init__(self):
        self._drivers: Dict[str, type] = {}  # driver_name -> driver_class
        self._cameras: Dict[str, ICamera] = {}  # camera_id -> camera_instance
        self._register_default_drivers()
    
    def register_driver(self, driver_name: str, driver_class: type) -> None:
        """注册相机驱动"""
        self._drivers[driver_name] = driver_class
    
    def discover_cameras(self) -> List[CameraInfo]:
        """发现所有可用相机"""
        cameras = []
        for driver_name, driver_class in self._drivers.items():
            try:
                driver = driver_class()
                cameras.extend(driver.discover())
            except Exception as e:
                logger.warning(f"Driver {driver_name} discovery failed: {e}")
        return cameras
    
    def create_camera(self, camera_info: CameraInfo) -> Optional[ICamera]:
        """创建相机实例"""
        driver_class = self._drivers.get(camera_info.driver_name)
        if not driver_class:
            return None
        
        camera = driver_class(camera_info)
        self._cameras[camera_info.camera_id] = camera
        return camera
    
    def get_camera(self, camera_id: str) -> Optional[ICamera]:
        """获取相机实例"""
        return self._cameras.get(camera_id)
    
    def remove_camera(self, camera_id: str) -> None:
        """移除相机实例"""
        if camera_id in self._cameras:
            camera = self._cameras[camera_id]
            if camera.is_connected():
                camera.disconnect()
            del self._cameras[camera_id]
    
    def _register_default_drivers(self) -> None:
        """注册默认驱动"""
        from core.camera.drivers.opencv_driver import OpenCVDriver
        self.register_driver("opencv", OpenCVDriver)

class OpenCVDriver(ICamera):
    """OpenCV相机驱动（支持USB摄像头）"""
    
    def __init__(self, camera_info: Optional[CameraInfo] = None):
        self._camera_info = camera_info
        self._cap: Optional[cv2.VideoCapture] = None
        self._is_connected = False
        self._is_capturing = False
        self._device_id = 0
        
        if camera_info:
            # 从camera_id解析device_id
            self._device_id = int(camera_info.camera_id.split("_")[-1])
    
    @staticmethod
    def discover() -> List[CameraInfo]:
        """发现所有USB摄像头"""
        cameras = []
        for i in range(10):  # 检测前10个设备
            cap = cv2.VideoCapture(i)
            if cap.isOpened():
                # 获取摄像头信息
                width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
                height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
                fps = cap.get(cv2.CAP_PROP_FPS)
                
                camera_info = CameraInfo(
                    camera_id=f"opencv_{i}",
                    name=f"USB Camera {i}",
                    model="Unknown",
                    serial_number=f"dev_{i}",
                    interface_type="usb",
                    driver_name="opencv"
                )
                cameras.append(camera_info)
                cap.release()
        return cameras
    
    def connect(self) -> bool:
        """连接相机"""
        try:
            self._cap = cv2.VideoCapture(self._device_id)
            self._is_connected = self._cap.isOpened()
            return self._is_connected
        except Exception as e:
            logger.error(f"Failed to connect camera: {e}")
            return False
    
    def disconnect(self) -> None:
        """断开连接"""
        if self._cap:
            self._cap.release()
            self._cap = None
        self._is_connected = False
        self._is_capturing = False
    
    def is_connected(self) -> bool:
        return self._is_connected
    
    def get_camera_info(self) -> CameraInfo:
        return self._camera_info
    
    def start_capture(self) -> bool:
        """开始采集（OpenCV无需显式开始）"""
        if not self._is_connected:
            return False
        self._is_capturing = True
        return True
    
    def stop_capture(self) -> None:
        """停止采集"""
        self._is_capturing = False
    
    def is_capturing(self) -> bool:
        return self._is_capturing
    
    def grab_frame(self, timeout_ms: int = 5000) -> Optional[np.ndarray]:
        """获取一帧图像"""
        if not self._is_connected or not self._cap:
            return None
        
        ret, frame = self._cap.read()
        if ret:
            return frame
        return None
    
    def list_parameters(self) -> List[str]:
        """列出OpenCV支持的参数"""
        return [
            "width", "height", "fps",
            "brightness", "contrast", "saturation", "hue",
            "exposure", "gain", "white_balance",
            "auto_exposure", "auto_white_balance",
            "focus", "auto_focus",
            "zoom", "pan", "tilt"
        ]
    
    def get_parameter_info(self, param_name: str) -> ParameterInfo:
        """获取参数信息"""
        # OpenCV参数映射
        param_map = {
            "width": ParameterInfo("width", "宽度", ParameterType.INTEGER, 0, 4096, 1, 640, unit="像素"),
            "height": ParameterInfo("height", "高度", ParameterType.INTEGER, 0, 4096, 1, 480, unit="像素"),
            "fps": ParameterInfo("fps", "帧率", ParameterType.FLOAT, 0, 120, 0.1, 30, unit="fps"),
            "brightness": ParameterInfo("brightness", "亮度", ParameterType.FLOAT, -1, 1, 0.01, 0),
            "contrast": ParameterInfo("contrast", "对比度", ParameterType.FLOAT, -1, 1, 0.01, 0),
            "saturation": ParameterInfo("saturation", "饱和度", ParameterType.FLOAT, -1, 1, 0.01, 0),
            "hue": ParameterInfo("hue", "色调", ParameterType.FLOAT, -1, 1, 0.01, 0),
            "exposure": ParameterInfo("exposure", "曝光时间", ParameterType.FLOAT, -13, -1, 0.1, -5, unit="log2(s)"),
            "gain": ParameterInfo("gain", "增益", ParameterType.FLOAT, 0, 100, 0.1, 0, unit="dB"),
            "white_balance": ParameterInfo("white_balance", "白平衡", ParameterType.INTEGER, 2000, 10000, 100, 4000, unit="K"),
            "auto_exposure": ParameterInfo("auto_exposure", "自动曝光", ParameterType.BOOLEAN, default_value=True),
            "auto_white_balance": ParameterInfo("auto_white_balance", "自动白平衡", ParameterType.BOOLEAN, default_value=True),
            "focus": ParameterInfo("focus", "对焦", ParameterType.FLOAT, 0, 100, 0.1, 0),
            "auto_focus": ParameterInfo("auto_focus", "自动对焦", ParameterType.BOOLEAN, default_value=True),
            "zoom": ParameterInfo("zoom", "缩放", ParameterType.FLOAT, 0, 100, 0.1, 0),
            "pan": ParameterInfo("pan", "平移", ParameterType.FLOAT, -180, 180, 0.1, 0, unit="度"),
            "tilt": ParameterInfo("tilt", "俯仰", ParameterType.FLOAT, -180, 180, 0.1, 0, unit="度"),
        }
        return param_map.get(param_name)
    
    def get_parameter(self, param_name: str) -> Any:
        """获取参数值"""
        if not self._cap:
            return None
        
        prop_map = {
            "width": cv2.CAP_PROP_FRAME_WIDTH,
            "height": cv2.CAP_PROP_FRAME_HEIGHT,
            "fps": cv2.CAP_PROP_FPS,
            "brightness": cv2.CAP_PROP_BRIGHTNESS,
            "contrast": cv2.CAP_PROP_CONTRAST,
            "saturation": cv2.CAP_PROP_SATURATION,
            "hue": cv2.CAP_PROP_HUE,
            "exposure": cv2.CAP_PROP_EXPOSURE,
            "gain": cv2.CAP_PROP_GAIN,
            "white_balance": cv2.CAP_PROP_WHITE_BALANCE_BLUE_U,
            "auto_exposure": cv2.CAP_PROP_AUTO_EXPOSURE,
            "auto_white_balance": cv2.CAP_PROP_AUTO_WB,
            "focus": cv2.CAP_PROP_FOCUS,
            "auto_focus": cv2.CAP_PROP_AUTOFOCUS,
            "zoom": cv2.CAP_PROP_ZOOM,
            "pan": cv2.CAP_PROP_PAN,
            "tilt": cv2.CAP_PROP_TILT,
        }
        
        prop = prop_map.get(param_name)
        if prop is not None:
            return self._cap.get(prop)
        return None
    
    def set_parameter(self, param_name: str, value: Any) -> bool:
        """设置参数值"""
        if not self._cap:
            return False
        
        prop_map = {
            "width": cv2.CAP_PROP_FRAME_WIDTH,
            "height": cv2.CAP_PROP_FRAME_HEIGHT,
            "fps": cv2.CAP_PROP_FPS,
            "brightness": cv2.CAP_PROP_BRIGHTNESS,
            "contrast": cv2.CAP_PROP_CONTRAST,
            "saturation": cv2.CAP_PROP_SATURATION,
            "hue": cv2.CAP_PROP_HUE,
            "exposure": cv2.CAP_PROP_EXPOSURE,
            "gain": cv2.CAP_PROP_GAIN,
            "white_balance": cv2.CAP_PROP_WHITE_BALANCE_BLUE_U,
            "auto_exposure": cv2.CAP_PROP_AUTO_EXPOSURE,
            "auto_white_balance": cv2.CAP_PROP_AUTO_WB,
            "focus": cv2.CAP_PROP_FOCUS,
            "auto_focus": cv2.CAP_PROP_AUTOFOCUS,
            "zoom": cv2.CAP_PROP_ZOOM,
            "pan": cv2.CAP_PROP_PAN,
            "tilt": cv2.CAP_PROP_TILT,
        }
        
        prop = prop_map.get(param_name)
        if prop is not None:
            return self._cap.set(prop, value)
        return False
